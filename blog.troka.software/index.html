using System;
using System.Runtime.CompilerServices;
using System.Runtime.Intrinsics;
using System.Runtime.Intrinsics.X86;

namespace TROKASoftware.HighPerformance
{
    /// <summary>
    /// AVX2 & FMA hardware-accelerated Blade Tip-Timing signal kernel.
    /// Performs 4 double FMA operations concurrently inside registers.
    /// </summary>
    public sealed unsafe class BttSignalKernel
    {
        [MethodImpl(MethodImplOptions.AggressiveOptimization)]
        public static void ProcessSignalSIMD(
            double* signal, double* filter, double* output, int length)
        {
            int i = 0;
            // 256-bit SIMD registers load 4 doubles concurrently
            for (; i <= length - 4; i += 4)
            {
                Vector256<double> sig = Avx.LoadVector256(signal + i);
                Vector256<double> filt = Avx.LoadVector256(filter + i);
                Vector256<double> acc = Avx.LoadVector256(output + i);

                // Single CPU cycle execution of acc = (sig * filt) + acc
                Vector256<double> res = Fma.MultiplyAdd(sig, filt, acc);
                Avx.Store(output + i, res);
            }
            // Clean up remaining elements
            for (; i < length; i++)
            {
                output[i] += signal[i] * filter[i];
            }
        }
    }
}
