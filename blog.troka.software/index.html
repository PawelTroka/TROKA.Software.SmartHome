using System;
using System.Collections.Generic;

namespace Troka.Software.HighPerformance
{
    /// <summary>
    /// High-performance VWAP and Strike Price calculator.
    /// Calculates marginal price required to fill volume.
    /// </summary>
    public class VwapCalculator : IVwapCalculator
    {
        public VwapResult CalculateStrikePrice(
            ReadOnlySpan<OrderBookLevel> levels, 
            decimal targetVolume)
        {
            decimal accumulatedVolume = 0;
            decimal accumulatedCost = 0;
            decimal strikePrice = 0;

            for (int i = 0; i < levels.Length; i++)
            {
                ref readonly var level = ref levels[i];
                decimal volumeToTake = Math.Min(
                    level.Volume, 
                    targetVolume - accumulatedVolume
                );

                accumulatedCost += volumeToTake * level.Price;
                accumulatedVolume += volumeToTake;
                strikePrice = level.Price;

                if (accumulatedVolume >= targetVolume)
                {
                    return new VwapResult(
                        vwap: accumulatedCost / targetVolume,
                        strikePrice: strikePrice, // Limit Price
                        isFullyFilled: true
                    );
                }
            }
            return new VwapResult(accumulatedCost / accumulatedVolume, 
                                  strikePrice, false);
        }
    }
}
