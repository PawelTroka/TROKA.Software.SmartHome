using System;
using System.Runtime.CompilerServices;
using System.Runtime.InteropServices;
using System.Threading;
using System.Runtime.Intrinsics;

namespace TROKASoftware.HighPerformance
{
    /// <summary>
    /// Low-latency order dispatcher. Suspends GC cycles,
    /// busy-waits on volatile shared memory, and routes trades.
    /// </summary>
    public sealed unsafe class LowLatencyOrderDispatcher : IDisposable
    {
        private readonly ConcurrentValueSortedDictionary<long, OrderLevel> _orderBook;
        private readonly byte* _sharedMemoryBuffer;
        private readonly int _spinWaitCycles;
        private bool _inNoGCRegion;

        [MethodImpl(MethodImplOptions.AggressiveOptimization | MethodImplOptions.AggressiveInlining)]
        public void DispatchAggressively(long targetVolume, decimal limitPrice)
        {
            // Prevent Garbage Collector latency spikes during active hot path
            if (!_inNoGCRegion && GC.TryStartNoGCRegion(1024 * 1024 * 64, true))
            {
                _inNoGCRegion = true;
            }

            try
            {
                // Busy-wait polling raw shared memory ticks using hardware spin wait
                int spinAttempts = 0;
                while (Volatile.Read(ref _sharedMemoryBuffer[0]) == 0)
                {
                    if (++spinAttempts > _spinWaitCycles)
                    {
                        Thread.SpinWait(64); // Low-level CPU pause instruction
                        spinAttempts = 0;
                    }
                }

                // Lock-optimized retrieval from sorted concurrent structures
                ref var bestLevel = ref _orderBook.GetRef(limitPrice);
                
                // Execute direct zero-copy Aeron publishing or direct IPC transmission
                if (bestLevel.Volume >= targetVolume)
                {
                    TransmitDirect(_sharedMemoryBuffer + 8, sizeof(OrderRequest));
                }
            }
            finally
            {
                if (_inNoGCRegion)
                {
                    GC.EndNoGCRegion();
                    _inNoGCRegion = false;
                }
            }
        }
    }
}
